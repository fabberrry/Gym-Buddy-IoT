cmake_minimum_required(VERSION 3.16)
project(doorway_counter LANGUAGES CXX)
add_library(people_counter src/people_counter.cpp)
target_include_directories(people_counter PUBLIC include)
target_compile_features(people_counter PUBLIC cxx_std_17)
set_target_properties(people_counter PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(people_counter PRIVATE /W4)
else()
  target_compile_options(people_counter PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(counter_sim examples/simulate.cpp)
target_link_libraries(counter_sim PRIVATE people_counter)
enable_testing()
add_executable(counter_tests tests/counter_tests.cpp)
target_link_libraries(counter_tests PRIVATE people_counter)
add_test(NAME counter_tests COMMAND counter_tests)
add_library(room_state src/room_state.cpp)
target_link_libraries(room_state PUBLIC people_counter)
add_executable(people_counter_gateway apps/people_counter_gateway.cpp)
target_link_libraries(people_counter_gateway PRIVATE room_state)
add_executable(room_state_tests tests/room_state_tests.cpp)
target_link_libraries(room_state_tests PRIVATE room_state)
add_test(NAME room_state_tests COMMAND room_state_tests)
find_package(Python3 3.9 COMPONENTS Interpreter REQUIRED)
add_test(NAME desktop_integration COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_desktop.py $<TARGET_FILE:people_counter_gateway>)
